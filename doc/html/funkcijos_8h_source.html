143template<typename konteineris>
144void rikiavimas(konteineris& A, int kriterijus)
145{
146
147 auto byVardas = [](const Studentas& a, const Studentas& b)
148 {
149 return a.getVardas() < b.getVardas();
150 };
151 auto byPavarde = [](const Studentas& a, const Studentas& b)
152 {
153 return a.getPavarde() < b.getPavarde();
154 };
155 auto byGalVid = [](const Studentas& a, const Studentas& b)
156 {
157 return a.getGalutinisVid() < b.getGalutinisVid();
158 };
159 auto byGalMed = [](const Studentas& a, const Studentas& b)
160 {
161 return a.getGalutinisMed() < b.getGalutinisMed();
162 };
163
164 //Listo rikiavimas
165 if constexpr (std::is_same_v<konteineris, std::list<Studentas>>)
166 {
167 switch(kriterijus)
168 {
169 case 1: A.sort(byVardas); break;
170 case 2: A.sort(byPavarde); break;
171 case 3: A.sort(byGalVid); break;
172 case 4: A.sort(byGalMed); break;
173 }
174 }
175 //Vector arba deque rikiavimas
176 else
177 {
178 switch(kriterijus)
179 {
180 case 1: std::sort(A.begin(), A.end(), byVardas); break;
181 case 2: std::sort(A.begin(), A.end(), byPavarde); break;
182 case 3: std::sort(A.begin(), A.end(), byGalVid); break;
183 case 4: std::sort(A.begin(), A.end(), byGalMed); break;
184 }
185 }
